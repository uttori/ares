# Standalone C++ sink ownership/failure tests, without emulator or JS dependencies.
add_executable(gdb-capture-file file-test.cpp)
find_package(Threads REQUIRED)
target_include_directories(gdb-capture-file PRIVATE ${CMAKE_SOURCE_DIR}/nall)
target_link_libraries(gdb-capture-file PRIVATE Threads::Threads)
target_compile_definitions(gdb-capture-file PRIVATE ARES_CAPTURE_FILE_TEST)
set_target_properties(gdb-capture-file PROPERTIES FOLDER tests PREFIX "")
ares_configure_executable(gdb-capture-file)
add_test(NAME gdb-capture-file COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/file-test.sh $<TARGET_FILE:gdb-capture-file>)
set_tests_properties(gdb-capture-file PROPERTIES TIMEOUT 30)
