# Exercise the production resize method with deterministic monitors and a window surface.
set(source "${CMAKE_SOURCE_DIR}/desktop-ui/presentation/presentation.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" contents)
string(FIND "${contents}" "auto Presentation::resizeWindow(" begin)
string(FIND "${contents}" "auto Presentation::refreshWindowSizeMenu()" end)
if(begin LESS 0 OR end LESS_EQUAL begin)
  message(FATAL_ERROR "Cannot locate Presentation::resizeWindow")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${contents}" ${begin} ${length} method)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/resize.inc" CONTENT "${method}" @ONLY)
add_executable(presentation-resize presentation-resize.cpp)
target_include_directories(presentation-resize PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(presentation-resize PRIVATE ares::nall)
set_target_properties(presentation-resize PROPERTIES FOLDER tests PREFIX "")
target_enable_subproject(presentation-resize "Presentation resize tests")
set(CONSOLE TRUE)
ares_configure_executable(presentation-resize)
add_test(NAME presentation-resize COMMAND presentation-resize)
set_tests_properties(presentation-resize PROPERTIES TIMEOUT 5)
