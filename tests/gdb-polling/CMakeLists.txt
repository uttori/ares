# Compile the actual method against a deterministic transport, without a socket or wall-clock sleeps.
set(polling_source "${CMAKE_SOURCE_DIR}/nall/nall/gdb/server.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${polling_source}")
file(READ "${polling_source}" server_source)
string(FIND "${server_source}" "  auto Server::updateLoop() -> void {" method_begin)
string(FIND "${server_source}" "  auto Server::getStatusText(" method_end)
if(method_begin LESS 0 OR method_end LESS_EQUAL method_begin)
  message(FATAL_ERROR "Cannot locate Server::updateLoop in ${polling_source}")
endif()
math(EXPR method_length "${method_end} - ${method_begin}")
string(SUBSTRING "${server_source}" ${method_begin} ${method_length} polling_method)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/polling-loop.inc" CONTENT "${polling_method}" @ONLY)

add_executable(gdb-polling gdb-polling.cpp)
target_include_directories(gdb-polling PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(gdb-polling PRIVATE ares::nall)
set_target_properties(gdb-polling PROPERTIES FOLDER tests PREFIX "")
target_enable_subproject(gdb-polling "GDB polling-window tests")
set(CONSOLE TRUE)
ares_configure_executable(gdb-polling)

add_test(NAME gdb-polling-window COMMAND gdb-polling)
set_tests_properties(gdb-polling-window PROPERTIES TIMEOUT 5)
target_sources(gdb-polling PRIVATE run-tests.sh README.md)
source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES gdb-polling.cpp)
